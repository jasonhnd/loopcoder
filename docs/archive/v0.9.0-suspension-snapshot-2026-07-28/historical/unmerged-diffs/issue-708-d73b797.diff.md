+func validateChildPlanGraph(ctx context.Context, tx Tx, parent RunNode, children []RunNode, plan ChildPlanRecord, edges []RunEdgeRecord) error {
+	parent.RunID = strings.TrimSpace(parent.RunID)
+	parent.RootRunID = firstNonEmptyStorage(parent.RootRunID, parent.RunID)
+	plan.ParentRunID = strings.TrimSpace(plan.ParentRunID)
+	plan.RootRunID = strings.TrimSpace(plan.RootRunID)

+		return fmt.Errorf("persist child plan graph: parent_run_id and root_run_id are required")
+	}
+	if plan.ParentRunID != parent.RunID {
+		return fmt.Errorf("persist child plan graph: plan parent_run_id %q does not match parent run %q", plan.ParentRunID, parent.RunID)
+	}

+		return fmt.Errorf("persist child plan graph: parent root_run_id %q does not match plan root_run_id %q", parent.RootRunID, plan.RootRunID)
+	}

+		return fmt.Errorf("persist child plan graph: parent depth must be non-negative")
+	}
+	if parent.Depth == 0 && parent.RootRunID != parent.RunID {
+		return fmt.Errorf("persist child plan graph: root parent %q must use itself as root_run_id, got %q", parent.RunID, parent.RootRunID)
+	}
+
+	existingParent, parentExists, err := loadStoredRun(ctx, tx, parent.RunID)
+	if err != nil {
+		return err
+	}
+	if parent.Depth > 0 && !parentExists {
+		return fmt.Errorf("persist child plan graph: parent run %q at depth %d is missing from durable graph", parent.RunID, parent.Depth)
+	}

+		if err := validateCompatibleExistingRun("parent", existingParent, parent.RunID, "", parent.RootRunID, parent.Depth); err != nil {
+			return err
+		}
+		if existingParent.ParentRunID == parent.RunID {
+			return fmt.Errorf("persist child plan graph: parent run %q references itself as parent", parent.RunID)
+		}
+	}

+	ancestors, err := loadAncestorSet(ctx, tx, parent.RunID)
+	if err != nil {
+		return err
+	}

+		edge.ParentRunID = strings.TrimSpace(edge.ParentRunID)

+		}
+		if child.ParentRunID != parent.RunID {
+			return fmt.Errorf("persist child plan graph: child %q parent_run_id %q does not match %q", child.RunID, child.ParentRunID, parent.RunID)
+		}
+		if child.RootRunID != parent.RootRunID || edge.RootRunID != parent.RootRunID {
+			return fmt.Errorf("persist child plan graph: child %q root_run_id mismatch", child.RunID)
+		}
+		if child.Depth != parent.Depth+1 || edge.Depth != child.Depth {

+			return fmt.Errorf("persist child plan graph: child %q plan_id and child_key are required", child.RunID)
+		}
+		if child.RunID == parent.RunID {
+			return fmt.Errorf("persist child plan graph: child %q cannot reuse parent_run_id", child.RunID)
+		}

+		}

+	}

+func validateCompatibleExistingRun(label string, existing RunNode, runID, parentRunID, rootRunID string, depth int) error {
+	if existing.RunID != runID {
+		return fmt.Errorf("persist child plan graph: %s run lookup returned %q for %q", label, existing.RunID, runID)
+	}

+		return fmt.Errorf("persist child plan graph: %s run %q already belongs to parent %q, not %q", label, runID, existing.ParentRunID, parentRunID)
+	}
+	if strings.TrimSpace(existing.RootRunID) != "" && strings.TrimSpace(existing.RootRunID) != rootRunID {
+		return fmt.Errorf("persist child plan graph: %s run %q has root_run_id %q, not %q", label, runID, existing.RootRunID, rootRunID)
+	}
+	if existing.Depth != depth {
+		return fmt.Errorf("persist child plan graph: %s run %q has depth %d, not %d", label, runID, existing.Depth, depth)
+	}
+	return nil

+func loadAncestorSet(ctx context.Context, tx Tx, runID string) (map[string]bool, error) {
+	ancestors := map[string]bool{}
+	seen := map[string]bool{}
+	current := strings.TrimSpace(runID)
+	for current != "" {
+		if seen[current] {
+			return nil, fmt.Errorf("persist child plan graph: existing parent chain contains a cycle at %q", current)
+		}
+		seen[current] = true
+		run, ok, err := loadStoredRun(ctx, tx, current)
+		if err != nil {
+			return nil, err
+		}
+		if !ok || strings.TrimSpace(run.ParentRunID) == "" {
+			return ancestors, nil
+		}
+		parent := strings.TrimSpace(run.ParentRunID)
+		ancestors[parent] = true
+		current = parent
+	}
+	return ancestors, nil
+}

+func loadRunEdgeGraph(ctx context.Context, tx Tx) (map[string][]string, error) {
+	rows, err := tx.Query(ctx, `SELECT parent_run_id, child_run_id FROM run_edges`)
+	if err != nil {
+		return nil, fmt.Errorf("persist child plan graph: load run edges: %w", err)
+	}
+	defer rows.Close()
+	graph := map[string][]string{}
+	for rows.Next() {
+		var parent, child string
+		if err := rows.Scan(&parent, &child); err != nil {
+			return nil, fmt.Errorf("persist child plan graph: scan run edge: %w", err)
+		}
+		parent = strings.TrimSpace(parent)
+		child = strings.TrimSpace(child)
+		if parent != "" && child != "" {
+			graph[parent] = append(graph[parent], child)
+		}
+	}
+	if err := rows.Err(); err != nil {
+		return nil, fmt.Errorf("persist child plan graph: scan run edges: %w", err)
+	}
+	return graph, nil
+}

+func reachesRun(graph map[string][]string, start, target string) bool {
+	start = strings.TrimSpace(start)
+	target = strings.TrimSpace(target)
+	if start == "" || target == "" {
+		return false
+	}
+	seen := map[string]bool{}
+	stack := []string{start}
+	for len(stack) > 0 {
+		current := stack[len(stack)-1]
+		stack = stack[:len(stack)-1]
+		if current == target {
+			return true
+		}
+		if seen[current] {
+			continue
+		}
+		seen[current] = true
+		stack = append(stack, graph[current]...)
+	}
+	return false
+}

+	}
+	return ""

+func durableGraphConsistencyCheck(ctx context.Context, db *sql.DB) error {
+	runRows, err := db.QueryContext(ctx, `SELECT id, COALESCE(parent_run_id, ''), COALESCE(root_run_id, ''), depth FROM runs`)
+	if err != nil {
+		return fmt.Errorf("durable run graph: load runs: %w", err)
+	}
+	runs := map[string]durableGraphRun{}
+	for runRows.Next() {
+		var run durableGraphRun
+		if err := runRows.Scan(&run.id, &run.parent, &run.root, &run.depth); err != nil {
+			runRows.Close()
+			return fmt.Errorf("durable run graph: scan runs: %w", err)
+		}
+		runs[run.id] = run
+	}
+	if err := runRows.Close(); err != nil {
+		return fmt.Errorf("durable run graph: scan runs: %w", err)
+	}
+
+	edgeRows, err := db.QueryContext(ctx, `SELECT parent_run_id, child_run_id, root_run_id, plan_id, child_key, depth, ordinal, scope_json, permission, aggregation_json, status FROM run_edges`)
+	if err != nil {
+		return fmt.Errorf("durable run graph: load edges: %w", err)
+	}
+	var edges []durableGraphEdge
+	graph := map[string][]string{}
+	var problems []string
+	for edgeRows.Next() {
+		var edge durableGraphEdge
+		if err := edgeRows.Scan(&edge.parent, &edge.child, &edge.root, &edge.planID, &edge.childKey, &edge.depth, &edge.ordinal, &edge.scope, &edge.permission, &edge.aggregation, &edge.status); err != nil {
+			edgeRows.Close()
+			return fmt.Errorf("durable run graph: scan edges: %w", err)
+		}
+		edges = append(edges, edge)
+		graph[edge.parent] = append(graph[edge.parent], edge.child)
+		if edge.parent == edge.child {
+			problems = append(problems, fmt.Sprintf("self edge %s -> %s", edge.parent, edge.child))
+		}
+	}
+	if err := edgeRows.Close(); err != nil {
+		return fmt.Errorf("durable run graph: scan edges: %w", err)
+	}
+
+	for _, edge := range edges {
+		if edge.parent != edge.child && durableReachesRun(graph, edge.child, edge.parent) {
+			problems = append(problems, fmt.Sprintf("cycle involving %s -> %s", edge.parent, edge.child))
+			break
+		}
+		if !durableEdgeIsEnriched(edge) {
+			continue
+		}
+		parent, parentOK := runs[edge.parent]
+		child, childOK := runs[edge.child]
+		if !parentOK || !childOK {
+			continue
+		}
+		if strings.TrimSpace(child.parent) != "" && child.parent != edge.parent {
+			problems = append(problems, fmt.Sprintf("edge %s -> %s disagrees with child parent_run_id %s", edge.parent, edge.child, child.parent))
+		}
+		if strings.TrimSpace(edge.root) == "" || strings.TrimSpace(child.root) == "" || edge.root != child.root {
+			problems = append(problems, fmt.Sprintf("edge %s -> %s has inconsistent root_run_id", edge.parent, edge.child))
+		}
+		if strings.TrimSpace(parent.root) != "" && strings.TrimSpace(edge.root) != "" && parent.root != edge.root {
+			problems = append(problems, fmt.Sprintf("edge %s -> %s root_run_id does not match parent", edge.parent, edge.child))
+		}
+		if edge.depth != child.depth {
+			problems = append(problems, fmt.Sprintf("edge %s -> %s depth %d disagrees with child depth %d", edge.parent, edge.child, edge.depth, child.depth))
+		}
+	}
+	if len(problems) > 0 {
+		sort.Strings(problems)
+		detail := strings.Join(problems, "; ")
+		if len(problems) > 3 {
+			detail = strings.Join(problems[:3], "; ") + fmt.Sprintf("; plus %d more", len(problems)-3)
+		}
+		return fmt.Errorf("durable run graph has %d problem(s): %s", len(problems), detail)
+	}
+	return nil
+}

+func durableEdgeIsEnriched(edge durableGraphEdge) bool {
+	return strings.TrimSpace(edge.root) != "" ||
+		strings.TrimSpace(edge.planID) != "" ||
+		strings.TrimSpace(edge.childKey) != "" ||
+		strings.TrimSpace(edge.scope) != "" && strings.TrimSpace(edge.scope) != "{}" ||
+		strings.TrimSpace(edge.permission) != "" ||
+		strings.TrimSpace(edge.aggregation) != "" && strings.TrimSpace(edge.aggregation) != "{}" ||
+		strings.TrimSpace(edge.status) != ""

+func durableReachesRun(graph map[string][]string, start, target string) bool {
+	seen := map[string]bool{}
+	stack := []string{start}
+	for len(stack) > 0 {
+		current := stack[len(stack)-1]
+		stack = stack[:len(stack)-1]
+		if current == target {
+			return true
+		}
+		if seen[current] {
+			continue
+		}
+		seen[current] = true
+		stack = append(stack, graph[current]...)
+	}
+	return false
+}
